import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# ---------- 1. LOAD & PRE-PROCESS ----------
df = pd.read_csv("sales_data_sample.csv", encoding="latin1")   # file is not UTF-8
print(df.shape)

cols = ["QUANTITYORDERED", "PRICEEACH", "SALES", "MSRP"]      # numeric columns to cluster on
data = df[cols].dropna()
print("Missing values:", df[cols].isna().sum().sum())

X = StandardScaler().fit_transform(data)   # K-Means uses distances, so scale first

# ---------- 2. ELBOW METHOD ----------
inertia = []
for k in range(1, 11):
    inertia.append(KMeans(n_clusters=k, n_init=10, random_state=42).fit(X).inertia_)

plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("Number of clusters (k)")
plt.ylabel("Inertia (within-cluster sum of squares)")
plt.title("Elbow method")
plt.show()

# ---------- 3. K-MEANS WITH CHOSEN k ----------
k = 3   # <- set this to the "elbow" point you see in the plot
data = data.copy()
data["Cluster"] = KMeans(n_clusters=k, n_init=10, random_state=42).fit_predict(X)

print("\nCluster sizes:\n", data["Cluster"].value_counts().sort_index())
print("\nCluster averages:\n", data.groupby("Cluster").mean().round(2))

plt.scatter(data["QUANTITYORDERED"], data["SALES"], c=data["Cluster"], cmap="viridis", s=10)
plt.xlabel("QUANTITYORDERED")
plt.ylabel("SALES")
plt.title(f"K-Means clusters (k={k})")
plt.show()
